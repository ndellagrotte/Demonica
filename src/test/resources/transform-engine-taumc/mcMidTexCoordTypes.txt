    _draw_id = (packed_draw_params >> 8) & 0xFFu;
    _vert_tex_light_coord = ivec2((uvec2((a_LightCoord >> 16) & 0xFFFFu) >> uvec2(0, 8)) & uvec2(0xFFu));
}

#else

in vec3 a_PosId;
