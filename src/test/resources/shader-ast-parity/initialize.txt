==== a second declarator and a flat declaration

in vec3 position ; 
in vec4 vaColor ; 
out float mat , recolor ; 
flat out float isMoon ; 
out vec3 normals [ 2 ] ; 
out vec3 [ 2 ] tangents ; 
out vec2 texCoord ; 
out vec4 tintOut ; 
layout ( location = 3 ) out vec4 extra ; 
uniform mat4 mvp ; 
void main ( ) { 
	 isMoon = 0.0f ; 
	recolor = 0.0f ; 
	mat = 1.0 ; 
	texCoord = position . xy ; 
	tintOut = vaColor ; 
	normals [ 0 ] = position ; 
	gl_Position = mvp * vec4 ( position , 1.0 ) ; 
}
